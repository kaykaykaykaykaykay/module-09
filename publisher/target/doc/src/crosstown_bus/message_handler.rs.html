6pub trait MessageHandler<T> {
7    fn handle(&self, message: Box<T>) -> Result<(), HandleError> 
8        where T: Clone + BorshDeserialize + BorshSerialize + 'static;
9}
10
11#[derive(Debug)]
12pub struct HandleError {
13    details: String,
14    pub requeue: bool
15}
16
17impl HandleError {
18    pub fn new(details: String, requeue: bool) -> Self {
19        Self { details, requeue }
20    }
21}
22
23impl Error for HandleError {
24    fn description(&self) -> &str {
25        &self.details
26    }
27}
28
29impl fmt::Display for HandleError {
30    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
31        write!(f,"{}/requeue:{}",self.details, self.requeue)
32    }
33}

35pub(crate) fn send_message_to_handler<T>(delivery: amiquip::Delivery, handler: &Arc<impl MessageHandler<T> + Send + Sync>, channel: &amiquip::Channel) 
36    where T : BorshDeserialize + BorshSerialize + Clone + 'static {
37    let str_message = String::from_utf8_lossy(&delivery.body).to_string();
38    let mut buf = str_message.as_bytes();
39    if let Ok(model) = BorshDeserialize::deserialize(&mut buf) {
40        if let Err(err) = handler.handle(model) {
41            println!("{err}");
42            _ = delivery.nack(channel, err.requeue);
43        } else {
44            _ = delivery.ack(channel);
45        }
46    } else {
47        _ = delivery.nack(channel, false);
48        eprintln!("[crosstown_bus] Error trying to desserialize. Check message format. Message: {:?}", str_message);
49    }
50}
