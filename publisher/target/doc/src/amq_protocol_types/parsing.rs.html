7/// Parse the [AMQPValue](../type.AMQPValue.html) of the given [AMQPType](../type.AMQPType.html)
8pub fn parse_raw_value<'a>(i: &'a [u8], amqp_type: &AMQPType) -> Result<(&'a [u8], AMQPValue), nom::Err<&'a [u8]>> {
9    match *amqp_type {
10        AMQPType::Boolean        => map!(i, call!(parse_boolean),          AMQPValue::Boolean),
11        AMQPType::ShortShortInt  => map!(i, call!(parse_short_short_int),  AMQPValue::ShortShortInt),
12        AMQPType::ShortShortUInt => map!(i, call!(parse_short_short_uint), AMQPValue::ShortShortUInt),
13        AMQPType::ShortInt       => map!(i, call!(parse_short_int),        AMQPValue::ShortInt),
14        AMQPType::ShortUInt      => map!(i, call!(parse_short_uint),       AMQPValue::ShortUInt),
15        AMQPType::LongInt        => map!(i, call!(parse_long_int),         AMQPValue::LongInt),
16        AMQPType::LongUInt       => map!(i, call!(parse_long_uint),        AMQPValue::LongUInt),
17        AMQPType::LongLongInt    => map!(i, call!(parse_long_long_int),    AMQPValue::LongLongInt),
18        /* Rabbitmq treats LongLongUInt as a LongLongInt hence expose it as such */
19        AMQPType::LongLongUInt   => map!(i, call!(parse_long_long_int),    AMQPValue::LongLongInt),
20        AMQPType::Float          => map!(i, call!(parse_float),            AMQPValue::Float),
21        AMQPType::Double         => map!(i, call!(parse_double),           AMQPValue::Double),
22        AMQPType::DecimalValue   => map!(i, call!(parse_decimal_value),    AMQPValue::DecimalValue),
23        /* ShortString is only for internal usage and AMQPValue::ShortString shouldn't exist, hence return it as LongString */
24        AMQPType::ShortString    => map!(i, call!(parse_short_string),     AMQPValue::LongString),
25        AMQPType::LongString     => map!(i, call!(parse_long_string),      AMQPValue::LongString),
26        AMQPType::FieldArray     => map!(i, call!(parse_field_array),      AMQPValue::FieldArray),
27        AMQPType::Timestamp      => map!(i, call!(parse_timestamp),        AMQPValue::Timestamp),
28        AMQPType::FieldTable     => map!(i, call!(parse_field_table),      AMQPValue::FieldTable),
29        AMQPType::ByteArray      => map!(i, call!(parse_byte_array),       AMQPValue::ByteArray),
30        AMQPType::Void           => value!(i, AMQPValue::Void),
31    }
32}

63/// Parse the [AMQPFlags](../type.AMQPFlags.html) for which the names are provided
64pub fn parse_flags<'a, 'b>(i: &'a [u8], names: &'b [&'b str]) -> Result<(&'a [u8], AMQPFlags), nom::Err<&'a [u8]>> {
65    map!(i, take!((names.len() + 7)/8), |b: &[u8]| AMQPFlags::from_bytes(names, b))
66}
