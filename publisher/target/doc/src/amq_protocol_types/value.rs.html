6/// Enumeration referencing the possible AMQP values depending on the types
7#[derive(Clone, Debug, PartialEq, Deserialize, Serialize)]
8pub enum AMQPValue {
9    /// A bool
10    Boolean(Boolean),
11    /// An i8
12    ShortShortInt(ShortShortInt),
13    /// A u8
14    ShortShortUInt(ShortShortUInt),
15    /// An i16
16    ShortInt(ShortInt),
17    /// A u16
18    ShortUInt(ShortUInt),
19    /// An i32
20    LongInt(LongInt),
21    /// A u32
22    LongUInt(LongUInt),
23    /// An i64
24    LongLongInt(LongLongInt),
25    /// An f32
26    Float(Float),
27    /// An f64
28    Double(Double),
29    /// A decimal value
30    DecimalValue(DecimalValue),
31    /// A String
32    LongString(LongString),
33    /// An array of AMQPValue
34    FieldArray(FieldArray),
35    /// A timestamp (u32)
36    Timestamp(Timestamp),
37    /// A Map<String, AMQPValue>
38    FieldTable(FieldTable),
39    /// An array of bytes (RabbitMQ speicific)
40    ByteArray(ByteArray),
41    /// No value
42    Void,
43}
44
45impl AMQPValue {
46    /// Get the AMQPType of an AMQPValue
47    pub fn get_type(&self) -> AMQPType {
48        match *self {
49            AMQPValue::Boolean(_)        => AMQPType::Boolean,
50            AMQPValue::ShortShortInt(_)  => AMQPType::ShortShortInt,
51            AMQPValue::ShortShortUInt(_) => AMQPType::ShortShortUInt,
52            AMQPValue::ShortInt(_)       => AMQPType::ShortInt,
53            AMQPValue::ShortUInt(_)      => AMQPType::ShortUInt,
54            AMQPValue::LongInt(_)        => AMQPType::LongInt,
55            AMQPValue::LongUInt(_)       => AMQPType::LongUInt,
56            AMQPValue::LongLongInt(_)    => AMQPType::LongLongInt,
57            AMQPValue::Float(_)          => AMQPType::Float,
58            AMQPValue::Double(_)         => AMQPType::Double,
59            AMQPValue::DecimalValue(_)   => AMQPType::DecimalValue,
60            AMQPValue::LongString(_)     => AMQPType::LongString,
61            AMQPValue::FieldArray(_)     => AMQPType::FieldArray,
62            AMQPValue::Timestamp(_)      => AMQPType::Timestamp,
63            AMQPValue::FieldTable(_)     => AMQPType::FieldTable,
64            AMQPValue::ByteArray(_)      => AMQPType::ByteArray,
65            AMQPValue::Void              => AMQPType::Void,
66        }
67    }
68}

70impl From<Value> for AMQPValue {
71    fn from(v: Value) -> AMQPValue {
72        From::from(&v)
73    }
74}
75
76impl<'a> From<&'a Value> for AMQPValue {
77    fn from(v: &Value) -> AMQPValue {
78        match *v {
79            Value::Bool(ref b)   => AMQPValue::Boolean(*b),
80            Value::Number(ref n) => {
81                if n.is_u64() {
82                    AMQPValue::LongLongInt(n.as_u64().unwrap() as i64)
83                } else if n.is_i64() {
84                    AMQPValue::LongLongInt(n.as_i64().unwrap())
85                } else {
86                    AMQPValue::Double(n.as_f64().unwrap())
87                }
88            },
89            Value::String(ref s) => AMQPValue::LongString(s.clone()),
90            Value::Array(ref v)  => AMQPValue::FieldArray(v.iter().map(From::from).collect()),
91            Value::Object(ref o) => AMQPValue::FieldTable(o.iter().fold(FieldTable::new(), |mut table, (k, v)| {
92                table.insert(k.clone(), From::from(v));
93                table
94            })),
95            Value::Null          => AMQPValue::Void,
96        }
97    }
98}

100#[cfg(test)]
101mod test {
102    use super::*;
103
104    use serde_json::{map, Number};
105
106    #[test]
107    fn test_from_bool_value() {
108        assert_eq!(AMQPValue::from(Value::Bool(false)), AMQPValue::Boolean(false));
109        assert_eq!(AMQPValue::from(Value::Bool(true)),  AMQPValue::Boolean(true));
110    }
111
112    #[test]
113    fn test_from_number_value() {
114        assert_eq!(AMQPValue::from(Value::Number(Number::from(42))),                 AMQPValue::LongLongInt(42));
115        assert_eq!(AMQPValue::from(Value::Number(Number::from(-42))),                AMQPValue::LongLongInt(-42));
116        assert_eq!(AMQPValue::from(Value::Number(Number::from_f64(42.42).unwrap())), AMQPValue::Double(42.42));
117    }
118
119    #[test]
120    fn test_from_string_value() {
121        assert_eq!(AMQPValue::from(Value::String(String::new())),      AMQPValue::LongString(String::new()));
122        assert_eq!(AMQPValue::from(Value::String("test".to_string())), AMQPValue::LongString("test".to_string()));
123    }
124
125    #[test]
126    fn test_from_array_value() {
127        assert_eq!(AMQPValue::from(Value::Array(Vec::new())),        AMQPValue::FieldArray(FieldArray::new()));
128        assert_eq!(AMQPValue::from(Value::Array(vec![Value::Null])), AMQPValue::FieldArray(vec![AMQPValue::Void]));
129    }
130
131    #[test]
132    fn test_from_object_value() {
133        let mut value_map = map::Map::new();
134        let mut table     = FieldTable::new();
135
136        value_map.insert("test".to_string(), Value::Null);
137        table.insert("test".to_string(), AMQPValue::Void);
138
139        assert_eq!(AMQPValue::from(Value::Object(map::Map::new())), AMQPValue::FieldTable(FieldTable::new()));
140        assert_eq!(AMQPValue::from(Value::Object(value_map)),       AMQPValue::FieldTable(table));
141    }
142
143    #[test]
144    fn test_from_null_value() {
145        assert_eq!(AMQPValue::from(Value::Null), AMQPValue::Void);
146    }
