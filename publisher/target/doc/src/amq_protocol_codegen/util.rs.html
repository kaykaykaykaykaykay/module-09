1/// Convert input to camel case
2pub fn camel_case(name: &str) -> String {
3    let mut new_word = true;
4    name.chars().fold("".to_string(), |mut result, ch| {
5        if ch == '-' || ch == '_' || ch == ' ' {
6            new_word = true;
7            result
8        } else {
9            result.push(if new_word { ch.to_ascii_uppercase() } else { ch });
10            new_word = false;
11            result
12        }
13    })
14}

16/// Convert input to snake case
17/// For the purpose of the AMQP codegen usage, we also handle a few special cases:
18/// "type" becomes "type_"
19/// "return" becomes "return_"
20/// A word needs to be composed of at least two letters, this makes UInt become uint and not u_int
21pub fn snake_case(name: &str) -> String {
22    match name {
23        "type"   => "type_".to_string(),
24        "return" => "return_".to_string(),
25        name     => {
26            let mut new_word       = false;
27            let mut last_was_upper = false;
28            name.chars().fold("".to_string(), |mut result, ch| {
29                if ch == '-' || ch == '_' || ch == ' ' {
30                    new_word = true;
31                    result
32                } else {
33                    let uppercase = ch.is_uppercase();
34                    if new_word || (!last_was_upper && !result.is_empty() && uppercase) {
35                        result.push('_');
36                        new_word = false;
37                    }
38                    last_was_upper = uppercase;
39                    result.push(if uppercase { ch.to_ascii_lowercase() } else { ch });
40                    result
41                }
42            })
43        }
44    }
45}

64        assert_eq!(snake_case("FooBar"),         "foo_bar");
