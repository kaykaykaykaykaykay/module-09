3/// Payload for a publisher confirmation message (either an [ack](enum.Confirm.html#variant.Ack) or
4/// a [nack](enum.Confirm.html#variant.Nack)) from the server.
5#[derive(Debug, Clone, Copy, PartialEq)]
6pub struct ConfirmPayload {
7    /// The tag from the server. Tags are sequentially increasing integers beginning with
8    /// 1 (once publisher confirms [are
9    ///   enabled](struct.Channel.html#method.enable_publisher_confirms) on the channel.
10    pub delivery_tag: u64,
11
12    /// If true, the confirmation applies to all previously-unconfirmed messages with delivery tags
13    /// less than or equal to this payload's [`delivery_tag`](#structfield.delivery_tag).
14    pub multiple: bool,
15}
16
17/// A publisher confirmation message from the server.
18#[derive(Debug, Clone, Copy, PartialEq)]
19pub enum Confirm {
20    /// Acknowledgment that the server has received the message(s) described by the associated
21    /// payload. Note that acks do not necessarily imply that the messages have been handled by a
22    /// consumer, merely that they have been received by the server.
23    Ack(ConfirmPayload),
24
25    /// Notification that the message(s) described by the associated payload have been rejected.
26    Nack(ConfirmPayload),
27}
28
29/// Helper to smooth out of order and/or `multiple: true` publisher confirmation messages.
30///
31/// If publisher confirms are enabled, the server may confirm messages out of order and/or may
32/// confirm multiple messages with a single [`Confirm`](enum.Confirm.html). `ConfirmSmoother`
33/// exists to "smooth" server messages out into an always-increasing-by-one sequence of
34/// confirmation messages.
35///
36/// # Example
37///
38/// ```rust
39/// use amiquip::{Confirm, ConfirmSmoother};
40/// use crossbeam_channel::Receiver;
41///
42/// // assume we've published n messages and want to wait for them to be confirmed
43/// fn wait_for_publisher_confirms(n: usize, receiver: &Receiver<Confirm>) {
44///     // NOTE: a new smoother assumes we will be receiving messages starting with
45///     // delivery_tag = 1, so this method is only valid if called after the first n
46///     // publishes on a channel. We could take a &mut ConfirmSmoother to be called
47///     // multiple times in succession on the same channel.
48///     let mut smoother = ConfirmSmoother::new();
49///     let mut acked = 0;
50///     let mut nacked = 0;
51///     while acked + nacked < n {
52///         // get a confirmation from the server; this may be out of order or a confirm
53///         // for multiple messages in one payload
54///         let raw_confirm = match receiver.recv() {
55///             Ok(raw_confirm) => raw_confirm,
56///             Err(_) => {
57///                 // the I/O thread dropped the sending side; either an error has occurred
58///                 // or another thread of ours closed the connection; either way we'll
59///                 // stop waiting
60///                 return;
61///             }
62///         };
63///
64///         // feed the raw confirm into the smoother. Two notes:
65///         //   1. We must run the returned iterator to the end or risk missing confirms.
66///         //   2. The iterator may produce 0, 1, or multiple independent confirmations.
67///         //      They will all have multiple: false.
68///         for confirm in smoother.process(raw_confirm) {
69///             match confirm {
70///                 Confirm::Ack(_) => {
71///                     acked += 1;
72///                 }
73///                 Confirm::Nack(_) => {
74///                     // server rejected message; need to do something else to
75///                     // track which messages were rejected
76///                     nacked += 1;
77///                 }
78///             }
79///         }
80///     }
81/// }
82/// ```
83#[derive(Debug, Clone)]
84pub struct ConfirmSmoother {
85    expected: u64,
86    out_of_order: HashMap<u64, Confirm>,
87}
88
89impl Default for ConfirmSmoother {
90    fn default() -> ConfirmSmoother {
91        ConfirmSmoother::new()
92    }
93}
94
95impl ConfirmSmoother {
96    /// Create a new `ConfirmSmoother`. It expects the next (in absolute order) delivery tag
97    /// received from the server to be `1`.
98    pub fn new() -> ConfirmSmoother {
99        ConfirmSmoother::with_expected_delivery_tag(1)
100    }
101
102    /// Create a new `ConfirmSmoother`. It expects the next (in absolute order) delivery tag
103    /// received from the server to be `expected`.
104    pub fn with_expected_delivery_tag(expected: u64) -> ConfirmSmoother {
105        ConfirmSmoother {
106            expected,
107            out_of_order: HashMap::new(),
108        }
109    }
110
111    /// Process a confirmation message from the server. Returns an iterator; each item returned by
112    /// the iterator will be a single (i.e., `multiple: false`) [`Confirm`](enum.Confirm.html). You
113    /// _must_ run the iterator to its completion or risk missing confirmations; future calls to
114    /// `process` will not return an iterator that will repeat confirms that would have been
115    /// returned by a previously returned iterator (even if that earlier iterator was dropped
116    /// before it ran to completion).
117    ///
118    /// The returned iterator may have 0 items (if `confirm` is a non-multiple confirmation that is
119    /// later than the next expected delivery tag), 1 item (if `confirm` exactly matches our next
120    /// expected delivery tag and we had not previously seen the next tag), or multiple items (if
121    /// `confirm` is a `multiple: true` confirmation or we've previously seen out-of-order tags
122    /// that are next sequentially after `confirm`'s tag).
123    pub fn process<'a>(&'a mut self, confirm: Confirm) -> impl Iterator<Item = Confirm> + 'a {
124        match confirm {
125            Confirm::Ack(inner) => self.new_iter(inner, Confirm::Ack),
126            Confirm::Nack(inner) => self.new_iter(inner, Confirm::Nack),
127        }
128    }
129
130    fn new_iter<'a>(
131        &'a mut self,
132        payload: ConfirmPayload,
133        to_confirm: fn(ConfirmPayload) -> Confirm,
134    ) -> impl Iterator<Item = Confirm> + 'a {
135        Iter {
136            parent: self,
137            payload,
138            next: None,
139            to_confirm: move |tag| {
140                to_confirm(ConfirmPayload {
141                    delivery_tag: tag,
142                    multiple: false,
143                })
144            },
145            done: false,
146        }
147    }
148}

150struct Iter<'a, F: Fn(u64) -> Confirm> {
151    parent: &'a mut ConfirmSmoother,
152    payload: ConfirmPayload,
153    next: Option<Confirm>,
154    to_confirm: F,
155    done: bool,
156}
157
158impl<'a, F> Drop for Iter<'a, F>
159where
160    F: Fn(u64) -> Confirm,
161{
162    fn drop(&mut self) {
163        while !self.done {
164            let _ = self.next();
165        }
166    }
167}
168
169impl<'a, F> Iterator for Iter<'a, F>
170where
171    F: Fn(u64) -> Confirm,
172{
173    type Item = Confirm;
174
175    fn next(&mut self) -> Option<Confirm> {
176        if self.done {
177            return None;
178        }
179
180        let payload = self.payload;
181
182        if payload.delivery_tag == self.parent.expected {
183            // exact match - we'll return this tag, and set next to an out-of-order
184            // entry for the next tag if we had one
185            self.parent.expected += 1;
186            self.next = self.parent.out_of_order.remove(&self.parent.expected);
187            return Some((self.to_confirm)(payload.delivery_tag));
188        }
189
190        if payload.delivery_tag > self.parent.expected {
191            // tag is in the future; if it's "multiple", keep sending all tags in
192            // between where we are now and payload.delivery_tag
193            if payload.multiple {
194                let ret = (self.to_confirm)(self.parent.expected);
195                self.parent.expected += 1;
196                return Some(ret);
197            } else {
198                // if it's _not_ multiple, stash it away in out_of_order
199                self.parent.out_of_order.insert(
200                    payload.delivery_tag,
201                    (self.to_confirm)(payload.delivery_tag),
202                );
203                self.done = true;
204                return None;
205            }
206        }
207
208        match self.next.take() {
209            Some(next) => {
210                // self.next is Some() only if a previous call to next() hit the tag==expected
211                // case _and_ out_of_order held a confirm for the next expected tag
212                self.parent.expected += 1;
213                self.next = self.parent.out_of_order.remove(&self.parent.expected);
214                Some(next)
215            }
216            None => {
217                self.done = true;
218                None
219            }
220        }
221    }
222}

228    fn single(delivery_tag: u64, f: fn(ConfirmPayload) -> Confirm) -> Confirm {
229        f(ConfirmPayload {
230            delivery_tag,
231            multiple: false,
232        })
233    }
234    fn multiple(delivery_tag: u64, f: fn(ConfirmPayload) -> Confirm) -> Confirm {
235        f(ConfirmPayload {
236            delivery_tag,
237            multiple: true,
238        })
239    }

241    #[test]
242    fn simple_single() {
243        let mut flat = ConfirmSmoother::new();
244        let one = flat.process(single(1, Confirm::Ack));
245        let expected = vec![single(1, Confirm::Ack)];
246        assert_eq!(expected, one.collect::<Vec<_>>());
247    }
248
249    #[test]
250    fn simple_multiple() {
251        let mut flat = ConfirmSmoother::new();
252        let three = flat.process(multiple(3, Confirm::Ack));
253        let expected = (1..=3).map(|i| single(i, Confirm::Ack)).collect::<Vec<_>>();
254        assert_eq!(expected, three.collect::<Vec<_>>());
255    }
256
257    #[test]
258    fn single_then_single() {
259        let mut flat = ConfirmSmoother::new();
260        let empty = flat.process(single(3, Confirm::Ack));
261        assert_eq!(empty.count(), 0);
262        let empty = flat.process(single(2, Confirm::Ack));
263        assert_eq!(empty.count(), 0);
264        let two = flat.process(single(1, Confirm::Nack));
265        let expected = vec![
266            single(1, Confirm::Nack),
267            single(2, Confirm::Ack),
268            single(3, Confirm::Ack),
269        ];
270        assert_eq!(expected, two.collect::<Vec<_>>());
271    }
272
273    #[test]
274    fn redelivery() {
275        let mut flat = ConfirmSmoother::new();
276        let two = flat.process(multiple(2, Confirm::Ack));
277        assert_eq!(two.count(), 2);
278
279        // getting another confirm for 1 or 2 should do nothing since flattener already
280        // dispatched a confirm for 1 and 2
281        let empty = flat.process(single(2, Confirm::Ack));
282        assert_eq!(empty.count(), 0);
283        let empty = flat.process(single(1, Confirm::Nack));
284        assert_eq!(empty.count(), 0);
285    }

287    #[test]
288    fn single_single_multiple_single() {
289        let mut flat = ConfirmSmoother::new();
290        let empty = flat.process(single(5, Confirm::Nack));
291        assert_eq!(empty.count(), 0);
292        let empty = flat.process(single(3, Confirm::Nack));
293        assert_eq!(empty.count(), 0);
294        let three = flat.process(multiple(2, Confirm::Ack));
295        assert_eq!(
296            vec![
297                single(1, Confirm::Ack),
298                single(2, Confirm::Ack),
299                single(3, Confirm::Nack)

301            three.collect::<Vec<_>>()
302        );
303        let two = flat.process(single(4, Confirm::Ack));
304        assert_eq!(
305            vec![single(4, Confirm::Ack), single(5, Confirm::Nack),],
306            two.collect::<Vec<_>>()
307        );
308    }
