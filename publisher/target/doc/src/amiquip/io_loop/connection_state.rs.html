26// Clippy warns about ConnectionState::Steady being much larger than the other variants, but we
27// expect ConnectionState to be in the Steady case almost all the time.
28#[allow(clippy::large_enum_variant)]
29pub(super) enum ConnectionState {
30    Steady(Channel0Slot),
31    ServerClosing(ConnectionClose),
32    ClientException,
33    ClientClosed,
34}

57fn send<T: Send + Sync + 'static>(tx: &Sender<T>, item: T) -> Result<()> {
58    // See comment in ChannelSlot::new() about the bound size of the control
59    // channel. If we're sending to a consumer channel, they are not bounded
60    // and will not return Full.
61    match tx.try_send(item) {
62        Ok(()) => Ok(()),
63        Err(TrySendError::Full(_)) => {
64            error!("internal error - bounded channel is unexpectedly full");
65            FrameUnexpectedSnafu.fail()
66        }
67        Err(TrySendError::Disconnected(_)) => {
68            error!("internal error - channel client dropped without being disconnected");
69            EventLoopClientDroppedSnafu.fail()
70        }
71    }
72}

74// When we set up a return listener, it's just a crossbeam channel. If it gets dropped,
75// we don't want to error; just start discarding returned messages.
76fn try_send_return(slot: &mut ChannelSlot, return_: Return) {
77    let return_ = if let Some(tx) = &slot.return_handler {
78        match tx.try_send(return_) {
79            Ok(()) => return,
80            Err(TrySendError::Full(return_)) | Err(TrySendError::Disconnected(return_)) => {
81                slot.return_handler = None;
82                return_
83            }
84        }
85    } else {
86        return_
87    };
88    warn!("discarding returned data {:?}", return_);
89}

91// When we set up a pub confirm listener, it's just a crossbeam channel. If it gets dropped,
92// we don't want to error; just start discarding acks/nacks
93fn try_send_confirm(slot: &mut ChannelSlot, confirm: Confirm) {
94    let confirm = if let Some(tx) = &slot.pub_confirm_handler {
95        match tx.try_send(confirm) {
96            Ok(()) => return,
97            Err(TrySendError::Full(confirm)) | Err(TrySendError::Disconnected(confirm)) => {
98                slot.pub_confirm_handler = None;
99                confirm
100            }
101        }
102    } else {
103        confirm
104    };
105    warn!("discarding returned data {:?}", confirm);
106}

108// When we set up a blocked connection listener, it's just a crossbeam channel. If it gets
109// dropped, we don't want to error; just start discarding blocked notifications.
110fn try_send_blocked(slot: &mut Channel0Slot, note: ConnectionBlockedNotification) {
111    if let Some(tx) = &slot.blocked_tx {
112        match tx.try_send(note) {
113            Ok(()) => (),
114            Err(_) => {
115                slot.blocked_tx = None;
116            }
117        }
118    }
119}

121impl ConnectionState {
122    fn client_exception(
123        &mut self,
124        inner: &mut Inner,
125        reply_code: AMQPHardError,
126        reply_text: String,
127    ) -> Result<()> {
128        error!("{} - closing connection", reply_text);
129        let close = ConnectionClose {
130            reply_code: reply_code.get_id(),
131            reply_text,
132            class_id: 0,
133            method_id: 0,
134        };
135        inner.push_method(0, AmqpConnection::Close(close));
136        inner.seal_writes();
137        *self = ConnectionState::ClientException;
138        Ok(())
139    }
140
141    pub(super) fn process(&mut self, inner: &mut Inner, frame: AMQPFrame) -> Result<()> {
142        // bail out if we shouldn't be getting frames
143        let ch0_slot = match self {
144            ConnectionState::Steady(ch0_slot) => ch0_slot,
145            ConnectionState::ClientException => return Ok(()),
146            ConnectionState::ServerClosing(_) | ConnectionState::ClientClosed => {
147                return FrameUnexpectedSnafu.fail();
148            }
149        };
150
151        match frame {
152            // Server-sent heartbeat
153            AMQPFrame::Heartbeat(0) => {
154                // nothing to do here; IoLoop already updated heartbeat timer when it
155                // received data on the socket
156                debug!("received heartbeat");
157            }
158            // We never expect to see a protocl header (we send it to begin the connection)
159            // or a heartbeat on a non-0 channel.
160            AMQPFrame::ProtocolHeader | AMQPFrame::Heartbeat(_) => return FrameUnexpectedSnafu.fail(),
161            // Server-initiated connection close.
162            AMQPFrame::Method(0, AMQPClass::Connection(AmqpConnection::Close(close))) => {
163                inner.push_method(0, AmqpConnection::CloseOk(ConnectionCloseOk {}));
164                inner.seal_writes();
165                let reply_code = close.reply_code;
166                let message = close.reply_text.clone();
167                let make_err = || Error::ServerClosedConnection {
168                    code: reply_code,
169                    message: message.clone(),
170                };
171                *self = ConnectionState::ServerClosing(close);
172
173                for (_, mut slot) in inner.chan_slots.drain() {
174                    send(&slot.tx, Err(make_err()))?;
175                    for (_, tx) in slot.consumers.drain() {
176                        send(&tx, ConsumerMessage::ServerClosedConnection(make_err()))?;
177                    }
178                }
179            }
180            // Server ack for client-initiated connection close.
181            AMQPFrame::Method(0, AMQPClass::Connection(AmqpConnection::CloseOk(close_ok))) => {
182                ch0_slot
183                    .common
184                    .tx
185                    .send(Ok(ChannelMessage::Method(AMQPClass::Connection(
186                        AmqpConnection::CloseOk(close_ok),
187                    ))))
188                    .map_err(|_| Error::EventLoopClientDropped)?;
189                *self = ConnectionState::ClientClosed;
190
191                for (_, mut slot) in inner.chan_slots.drain() {
192                    send(&slot.tx, Err(Error::ClientClosedConnection))?;
193                    for (_, tx) in slot.consumers.drain() {
194                        send(&tx, ConsumerMessage::ClientClosedConnection)?;
195                    }
196                }
197            }
198            // Server is blocking publishes due to an alarm on its side (e.g., low mem)
199            AMQPFrame::Method(0, AMQPClass::Connection(AmqpConnection::Blocked(blocked))) => {
200                warn!("server has blocked connection; reason = {}", blocked.reason);
201                let note = ConnectionBlockedNotification::Blocked(blocked.reason);
202                try_send_blocked(ch0_slot, note);
203            }
204            // Server has unblocked publishes
205            AMQPFrame::Method(0, AMQPClass::Connection(AmqpConnection::Unblocked(_))) => {
206                warn!("server has unblocked connection");
207                let note = ConnectionBlockedNotification::Unblocked;
208                try_send_blocked(ch0_slot, note);
209            }
210            // Reject all other expected channel 0 methods
211            AMQPFrame::Method(0, other) => {
212                let text = format!("do not know how to handle channel 0 method {:?}", other);
213                self.client_exception(inner, AMQPHardError::NOTIMPLEMENTED, text)?;
214            }
215            // Reject content frames on channel 0.
216            AMQPFrame::Header(0, _, _) | AMQPFrame::Body(0, _) => {
217                let text = format!("received illegal channel 0 frame {:?}", frame);
218                self.client_exception(inner, AMQPHardError::NOTALLOWED, text)?;
219            }
220            // Server-initiated channel close.
221            AMQPFrame::Method(n, AMQPClass::Channel(AmqpChannel::Close(close))) => {
222                warn!("server closing channel {}: {:?}", n, close);
223                let mut slot = slot_remove(inner, n)?;
224                let make_err = || Error::ServerClosedChannel {
225                    channel_id: n,
226                    code: close.reply_code,
227                    message: close.reply_text.clone(),
228                };
229                send(&slot.tx, Err(make_err()))?;
230                for (_, tx) in slot.consumers.drain() {
231                    send(&tx, ConsumerMessage::ServerClosedChannel(make_err()))?;
232                }
233                inner.push_method(n, AmqpChannel::CloseOk(ChannelCloseOk {}));
234            }
235            // Server ack for client-initiated channel close.
236            AMQPFrame::Method(n, AMQPClass::Channel(AmqpChannel::CloseOk(close_ok))) => {
237                // Closing is inherently racy; if we and the server both send a Close at
238                // the same time, we might see the server Close and then get a CloseOk, but
239                // we will have removed the slot when we got the close. It is therefore not
240                // an error to get a CloseOk for a nonexistent slot, since the server is
241                // confirming that a channel is gone (and we don't have it anymore anyway).
242                if let Ok(mut slot) = slot_remove(inner, n) {
243                    send(
244                        &slot.tx,
245                        Ok(ChannelMessage::Method(AMQPClass::Channel(
246                            AmqpChannel::CloseOk(close_ok),
247                        ))),
248                    )?;
249                    for (_, tx) in slot.consumers.drain() {
250                        send(&tx, ConsumerMessage::ClientClosedChannel)?;
251                    }
252                }
253            }
254            // Server ack for consume request.
255            AMQPFrame::Method(n, AMQPClass::Basic(AmqpBasic::ConsumeOk(consume_ok))) => {
256                let consumer_tag = consume_ok.consumer_tag;
257                let slot = slot_get_mut(inner, n)?;
258                match slot.consumers.entry(consumer_tag.clone()) {
259                    Entry::Occupied(_) => {
260                        return DuplicateConsumerTagSnafu {
261                            channel_id: n,
262                            consumer_tag,
263                        }
264                        .fail();
265                    }
266                    Entry::Vacant(entry) => {
267                        let (tx, rx) = crossbeam_channel::unbounded();
268                        entry.insert(tx);
269                        send(&slot.tx, Ok(ChannelMessage::ConsumeOk(consumer_tag, rx)))?;
270                    }
271                }
272            }
273            // Server-initiated consumer cancel.
274            AMQPFrame::Method(n, AMQPClass::Basic(AmqpBasic::Cancel(cancel))) => {
275                let consumer_tag = cancel.consumer_tag;
276                let slot = slot_get_mut(inner, n)?;
277                if let Some(tx) = slot.consumers.remove(&consumer_tag) {
278                    send(&tx, ConsumerMessage::ServerCancelled)?;
279                }
280                if !cancel.nowait {
281                    inner.push_method(n, AmqpBasic::CancelOk(CancelOk { consumer_tag }));
282                }
283            }
284            // Server ack for client-initiated consumer cancel.
285            AMQPFrame::Method(n, AMQPClass::Basic(AmqpBasic::CancelOk(cancel_ok))) => {
286                let slot = slot_get_mut(inner, n)?;
287                let consumer = slot.consumers.remove(&cancel_ok.consumer_tag);
288                send(
289                    &slot.tx,
290                    Ok(ChannelMessage::Method(AMQPClass::Basic(
291                        AmqpBasic::CancelOk(cancel_ok),
292                    ))),
293                )?;
294                if let Some(tx) = consumer {
295                    send(&tx, ConsumerMessage::ClientCancelled)?;
296                }
297            }
298            // Server beginning delivery of content to a consumer.
299            AMQPFrame::Method(n, AMQPClass::Basic(AmqpBasic::Deliver(deliver))) => {
300                let slot = slot_get_mut(inner, n)?;
301                slot.collector.collect_deliver(deliver)?;
302            }
303            // Server beginning return of undeliverable content.
304            AMQPFrame::Method(n, AMQPClass::Basic(AmqpBasic::Return(return_))) => {
305                let slot = slot_get_mut(inner, n)?;
306                slot.collector.collect_return(return_)?;
307            }
308            // Server ack for get (message incoming).
309            AMQPFrame::Method(n, AMQPClass::Basic(AmqpBasic::GetOk(get_ok))) => {
310                let slot = slot_get_mut(inner, n)?;
311                slot.collector.collect_get(get_ok)?;
312            }
313            // Server ack for get (no message).
314            AMQPFrame::Method(n, AMQPClass::Basic(AmqpBasic::GetEmpty(_))) => {
315                let slot = slot_get(inner, n)?;
316                send(&slot.tx, Ok(ChannelMessage::GetOk(Box::new(None))))?;
317            }
318            // Server ack for publish (publisher confirmation)
319            AMQPFrame::Method(n, AMQPClass::Basic(AmqpBasic::Ack(ack))) => {
320                let slot = slot_get_mut(inner, n)?;
321                let confirm = ConfirmPayload {
322                    delivery_tag: ack.delivery_tag,
323                    multiple: ack.multiple,
324                };
325                try_send_confirm(slot, Confirm::Ack(confirm));
326            }
327            // Server nack for publish (publisher confirmation)
328            AMQPFrame::Method(n, AMQPClass::Basic(AmqpBasic::Nack(nack))) => {
329                let slot = slot_get_mut(inner, n)?;
330                let confirm = ConfirmPayload {
331                    delivery_tag: nack.delivery_tag,
332                    multiple: nack.multiple,
333                };
334                try_send_confirm(slot, Confirm::Nack(confirm));
335            }
336            // Generic ack messages we send back to the caller.
337            AMQPFrame::Method(n, method @ AMQPClass::Basic(AmqpBasic::QosOk(_)))
338            | AMQPFrame::Method(n, method @ AMQPClass::Basic(AmqpBasic::RecoverOk(_)))
339            | AMQPFrame::Method(n, method @ AMQPClass::Channel(AmqpChannel::OpenOk(_)))
340            | AMQPFrame::Method(n, method @ AMQPClass::Confirm(AmqpConfirm::SelectOk(_)))
341            | AMQPFrame::Method(n, method @ AMQPClass::Exchange(AmqpExchange::DeclareOk(_)))
342            | AMQPFrame::Method(n, method @ AMQPClass::Exchange(AmqpExchange::DeleteOk(_)))
343            | AMQPFrame::Method(n, method @ AMQPClass::Exchange(AmqpExchange::BindOk(_)))
344            | AMQPFrame::Method(n, method @ AMQPClass::Exchange(AmqpExchange::UnbindOk(_)))
345            | AMQPFrame::Method(n, method @ AMQPClass::Queue(AmqpQueue::DeclareOk(_)))
346            | AMQPFrame::Method(n, method @ AMQPClass::Queue(AmqpQueue::DeleteOk(_)))
347            | AMQPFrame::Method(n, method @ AMQPClass::Queue(AmqpQueue::BindOk(_)))
348            | AMQPFrame::Method(n, method @ AMQPClass::Queue(AmqpQueue::PurgeOk(_)))
349            | AMQPFrame::Method(n, method @ AMQPClass::Queue(AmqpQueue::UnbindOk(_))) => {
350                let slot = slot_get(inner, n)?;
351                trace!(
352                    "trying to send method to client for channel {}: {:?}",
353                    n,
354                    method
355                );
356                send(&slot.tx, Ok(ChannelMessage::Method(method)))?;
357            }
358            // Methods we do not handle
359            AMQPFrame::Method(n, method @ AMQPClass::Access(_))
360            | AMQPFrame::Method(n, method @ AMQPClass::Channel(AmqpChannel::Flow(_)))
361            | AMQPFrame::Method(n, method @ AMQPClass::Channel(AmqpChannel::FlowOk(_)))
362            | AMQPFrame::Method(n, method @ AMQPClass::Tx(_)) => {
363                let text = format!(
364                    "do not know how to handle channel {} method {:?}",
365                    n, method
366                );
367                self.client_exception(inner, AMQPHardError::NOTIMPLEMENTED, text)?;
368            }
369            // Methods that are illegal coming from the server
370            AMQPFrame::Method(n, method @ AMQPClass::Basic(AmqpBasic::Qos(_)))
371            | AMQPFrame::Method(n, method @ AMQPClass::Basic(AmqpBasic::Consume(_)))
372            | AMQPFrame::Method(n, method @ AMQPClass::Basic(AmqpBasic::Get(_)))
373            | AMQPFrame::Method(n, method @ AMQPClass::Basic(AmqpBasic::Publish(_)))
374            | AMQPFrame::Method(n, method @ AMQPClass::Basic(AmqpBasic::Recover(_)))
375            | AMQPFrame::Method(n, method @ AMQPClass::Basic(AmqpBasic::RecoverAsync(_)))
376            | AMQPFrame::Method(n, method @ AMQPClass::Basic(AmqpBasic::Reject(_)))
377            | AMQPFrame::Method(n, method @ AMQPClass::Channel(AmqpChannel::Open(_)))
378            | AMQPFrame::Method(n, method @ AMQPClass::Confirm(AmqpConfirm::Select(_)))
379            | AMQPFrame::Method(n, method @ AMQPClass::Connection(_))
380            | AMQPFrame::Method(n, method @ AMQPClass::Exchange(AmqpExchange::Declare(_)))
381            | AMQPFrame::Method(n, method @ AMQPClass::Exchange(AmqpExchange::Delete(_)))
382            | AMQPFrame::Method(n, method @ AMQPClass::Exchange(AmqpExchange::Bind(_)))
383            | AMQPFrame::Method(n, method @ AMQPClass::Exchange(AmqpExchange::Unbind(_)))
384            | AMQPFrame::Method(n, method @ AMQPClass::Queue(AmqpQueue::Declare(_)))
385            | AMQPFrame::Method(n, method @ AMQPClass::Queue(AmqpQueue::Delete(_)))
386            | AMQPFrame::Method(n, method @ AMQPClass::Queue(AmqpQueue::Bind(_)))
387            | AMQPFrame::Method(n, method @ AMQPClass::Queue(AmqpQueue::Purge(_)))
388            | AMQPFrame::Method(n, method @ AMQPClass::Queue(AmqpQueue::Unbind(_))) => {
389                let text = format!("illegal channel {} method {:?}", n, method);
390                self.client_exception(inner, AMQPHardError::NOTALLOWED, text)?;
391            }
392            // Server sending content header as part of a deliver.
393            AMQPFrame::Header(n, _, header) => {
394                let slot = slot_get_mut(inner, n)?;
395                if let Some(collected) = slot.collector.collect_header(*header)? {
396                    match collected {
397                        CollectorResult::Delivery((consumer_tag, delivery)) => {
398                            let tx =
399                                slot.consumers
400                                    .get(&consumer_tag)
401                                    .context(UnknownConsumerTagSnafu {
402                                        channel_id: n,
403                                        consumer_tag,
404                                    })?;
405                            send(tx, ConsumerMessage::Delivery(delivery))?;
406                        }
407                        CollectorResult::Return(return_) => {
408                            try_send_return(slot, return_);
409                        }
410                        CollectorResult::Get(get) => {
411                            send(&slot.tx, Ok(ChannelMessage::GetOk(Box::new(Some(get)))))?;
412                        }
413                    }
414                }
415            }
416            // Server sending content body as part of a deliver.
417            AMQPFrame::Body(n, body) => {
418                let slot = slot_get_mut(inner, n)?;
419                if let Some(collected) = slot.collector.collect_body(body)? {
420                    match collected {
421                        CollectorResult::Delivery((consumer_tag, delivery)) => {
422                            let tx =
423                                slot.consumers
424                                    .get(&consumer_tag)
425                                    .context(UnknownConsumerTagSnafu {
426                                        channel_id: n,
427                                        consumer_tag,
428                                    })?;
429                            send(tx, ConsumerMessage::Delivery(delivery))?;
430                        }
431                        CollectorResult::Return(return_) => {
432                            try_send_return(slot, return_);
433                        }
434                        CollectorResult::Get(get) => {
435                            send(&slot.tx, Ok(ChannelMessage::GetOk(Box::new(Some(get)))))?;
436                        }
437                    }
438                }
439            }
440        }
441        Ok(())
442    }
