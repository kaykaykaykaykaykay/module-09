138    fn send(&mut self, message: IoLoopMessage) -> Result<()> {
139        self.tx
140            .send(message)
141            .map_err(|_| self.check_recv_for_error())
142    }
143
144    fn recv(&mut self) -> Result<ChannelMessage> {
145        self.rx.recv().map_err(|_| Error::EventLoopDropped)?
146    }
147
148    fn check_recv_for_error(&mut self) -> Error {
149        // failed to send to the I/O thread; possible causes are:
150        //   1. Server closed channel; we should see if there's a relevant message
151        //      waiting for us on rx.
152        //   2. I/O loop is actually gone.
153        // In either case, recv() will return Err. If it doesn't, we got somehow
154        // got a frame after a send failure - this should be impossible, but return
155        // FrameUnexpected just in case.
156        match self.recv() {
157            Ok(_) => {
158                error!("internal error - received unexpected frame after I/O thread disappeared");
159                Error::FrameUnexpected
160            }
161            Err(err) => err,
162        }
163    }
