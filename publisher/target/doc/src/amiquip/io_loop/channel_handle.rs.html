19// Each frame has 8 bytes of overhead (7 byte header, 1 byte frame-end), so when
20// we break our frames up into frame_max pieces, we need to account for this many
21// bytes.
22const FRAME_OVERHEAD: usize = 8;
23
24#[derive(Debug)]
25pub(crate) struct Channel0Handle {
26    handle: IoLoopHandle0,
27    frame_max: usize,
28}

30impl Channel0Handle {
31    pub(super) fn new(handle: IoLoopHandle0, mut frame_max: usize) -> Channel0Handle {
32        assert!(
33            handle.channel_id() == 0,
34            "handle for Channel0 must be channel 0"
35        );
36        if frame_max == 0 {
37            frame_max = usize::max_value();
38        }
39        frame_max -= FRAME_OVERHEAD;
40        Channel0Handle { handle, frame_max }
41    }

50    pub(crate) fn close_connection(&mut self) -> Result<()> {
51        let close = ConnectionClose {
52            reply_code: u16::from(REPLY_SUCCESS),
53            reply_text: "goodbye".to_string(),
54            class_id: 0,
55            method_id: 0,
56        };

60    pub(crate) fn open_channel(&mut self, channel_id: Option<u16>) -> Result<ChannelHandle> {
61        let mut handle = self.handle.allocate_channel(channel_id)?;
62
63        debug!("opening channel {}", handle.channel_id());
64        let out_of_band = String::new();
65        let open = AmqpChannel::Open(ChannelOpen { out_of_band });
66
67        let open_ok = handle.call::<_, ChannelOpenOk>(open)?;
68        trace!("got open-ok: {:?}", open_ok);
69        Ok(ChannelHandle {
70            handle,
71            frame_max: self.frame_max,
72        })
73    }

153    pub(crate) fn send_content(
154        &mut self,
155        mut content: &[u8],
156        class_id: u16,
157        properties: &AMQPProperties,
158    ) -> Result<()> {
159        trace!(
160            "sending content header on channel {} (class_id = {}, len = {})",
161            self.channel_id(),
162            class_id,
163            content.len()
164        );
165        self.handle
166            .send_content_header(class_id, content.len(), properties)?;
167
168        while content.len() > self.frame_max {
169            trace!(
170                "sending partial content body frame on channel {} (len = {})",
171                self.channel_id(),
172                self.frame_max
173            );
174            self.handle.send_content_body(&content[..self.frame_max])?;
175            content = &content[self.frame_max..];
176        }
177        if !content.is_empty() {
178            trace!(
179                "sending final content body frame on channel {} (len = {})",
180                self.channel_id(),
181                content.len()
182            );
183            self.handle.send_content_body(content)?;
184        }
185        Ok(())
186    }
